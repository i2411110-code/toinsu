/** * news.js - 보험가온포탈 통합 뉴스 가동 엔진 * index.html 내장 전산 뷰 테두리 및 탭, 지표들을 실시간으로 정밀 결합 처리합니다. */ const NewsWidget = (() => { const clean = (str = '') => str.replace(//g, '').replace(/<\/b>/g, '') .replace(/"/g, '"').replace(/&/g, '&') .replace(/</g, '<').replace(/>/g, '>'); const fmtDate = (str) => { const d = new Date(str); return isNaN(d.getTime()) ? '' : d.toLocaleString('ko-KR', { month: '2-digit', day: '2-digit', hour: '2-digit', minute: '2-digit' }); }; const weatherEmoji = (desc = '') => desc.includes('맑') ? '☀️' : desc.includes('구름') || desc.includes('흐림') ? '☁️' : desc.includes('비') ? '🌧️' : desc.includes('눈') ? '❄️' : '☀️'; const api = { news: async (query, display = 10) => { try { const r = await fetch(`/api/naver-news?query=${encodeURIComponent(query)}&display=${display}`, { cache: 'no-store' }); const j = await r.json(); return j.items || []; } catch { return []; } }, exchange: async () => { try { const r = await fetch('/api/naver-exchange', { cache: 'no-store' }); const j = await r.json(); return j.items || []; } catch { return []; } }, market: async () => { try { const r = await fetch('/api/market', { cache: 'no-store' }); const j = await r.json(); return j.items || []; } catch { return []; } }, weather: async (region) => { try { const r = await fetch(`/api/weather?region=${region}`, { cache: 'no-store' }); return await r.json(); } catch { return { error: true, region, temp: 23, description: '맑음' }; } } }; // 실시간 데이터 파이프라인 출력 처리 const renderAllComponents = async (categoryKeyword = '전체 뉴스') => { let query = "보험 금융 경제"; if (categoryKeyword !== '전체 뉴스' && categoryKeyword !== '전체') { query = categoryKeyword; } const loaderTarget = document.getElementById('news-live-feed-target-container'); if(loaderTarget) loaderTarget.innerHTML = `
로딩 중...
`; // 병렬 데이터 패치 가동 const [newsItems, fxItems, mktItems] = await Promise.all([ api.news(query, 8), api.exchange(), api.market() ]); // 1. 좌측 MORNING NEWS 지면 매핑 const paperHeadline = document.getElementById('paper-main-headline-body'); if (paperHeadline && newsItems.length > 0) { paperHeadline.innerHTML = ` "${clean(newsItems[0].title)}" 발행처: 네이버 뉴스 · 스크랩: ${fmtDate(newsItems[0].pubDate)} ${clean(newsItems[0].description)} `; } // 2. 신문 상단 미니 지표 전산 연동 const miniIndexBox = document.getElementById('paper-mini-index-box'); if (miniIndexBox) { const usd = fxItems.find(i => i.label === 'USD') || { value: 1514.60, change: 5.20, direction: 'down' }; const kospi = mktItems.find(i => i.label === '코스피') || { value: 8545.98, change: 422.36, direction: 'up' }; const kosdaq = mktItems.find(i => i.label === '코스닥') || { value: 1034.03, change: 14.98, direction: 'up' }; miniIndexBox.innerHTML = `
USD / KRW
${Number(usd.value).toFixed(2)}
${usd.direction === 'up' ? '▲' : '▼'} ${Math.abs(usd.change).toFixed(2)}
KOSPI
${Number(kospi.value).toLocaleString('ko-KR', {maximumFractionDigits:2})}
${kospi.direction === 'up' ? '▲' : '▼'} ${Math.abs(kospi.change).toFixed(2)}
KOSDAQ
${Number(kosdaq.value).toLocaleString('ko-KR', {maximumFractionDigits:2})}
${kosdaq.direction === 'up' ? '▲' : '▼'} ${Math.abs(kosdaq.change).toFixed(2)}
 `; } // 3. 우측 실시간 카드 속보 피드 매핑 가동 if (loaderTarget) { loaderTarget.innerHTML = ''; if (newsItems.length <= 1) { loaderTarget.innerHTML = `
표시할 라이브 뉴스가 없습니다.
`; } else { const restFeed = newsItems.slice(1, 5); restFeed.forEach((item, idx) => { const isTop = idx === 0; const html = `
${isTop ? '주요속보' : '속보'} ${fmtDate(item.pubDate)}

${clean(item.title)}

${clean(item.description)}

기사 원문 보기
`; loaderTarget.insertAdjacentHTML('beforeend', html); }); } } // 4. 하단 실시간 종합 전산 금융판 매핑 가동 const bottomGrid = document.getElementById('news-bottom-market-grid'); if (bottomGrid) { const usd = fxItems.find(i => i.label === 'USD') || { value: 1512.40, change: 1.40, direction: 'up' }; const jpy = fxItems.find(i => i.label === 'JPY') || { value: 943.63, change: 1.78, direction: 'up' }; const kospi = mktItems.find(i => i.label === '코스피') || { value: 8864.24, change: 137.84, direction: 'up' }; const gold = mktItems.find(i => i.label === '국내 금 (원/g)') || { value: 209700, change: 80, direction: 'up' }; bottomGrid.innerHTML = `
USD / KRW (원/달러 환율)
${Number(usd.value).toFixed(2)}
${usd.direction === 'up' ? '▲' : '▼'} ${Math.abs(usd.change).toFixed(2)} (${usd.direction === 'up' ? '상승' : '하락'})
JPY / KRW (100엔 환율)
${Number(jpy.value).toFixed(2)}
${jpy.direction === 'up' ? '▲' : '▼'} ${Math.abs(jpy.change).toFixed(2)} (${jpy.direction === 'up' ? '상승' : '하락'})
KOSPI 종합지수
${Number(kospi.value).toLocaleString('ko-KR', {maximumFractionDigits:2})}
${kospi.direction === 'up' ? '▲' : '▼'} ${Math.abs(kospi.change).toFixed(2)} (${kospi.direction === 'up' ? '상승' : '하락'})
국내 금시세 (g당)
${Number(gold.value).toLocaleString('ko-KR', {maximumFractionDigits:0})}
${gold.direction === 'up' ? '▲' : '▼'} ${Math.abs(gold.change).toLocaleString('ko-KR', {maximumFractionDigits:0})} (${gold.direction === 'up' ? '상승' : '하락'})
 `; } }; const init = async () => { // 1. 전국 주요 도시 실시간 기상청 스크롤 바인딩 const regions = ['서울', '부산', '대구', '인천', '광주', '대전', '울산', '세종', '제주']; Promise.all(regions.map(r => api.weather(r))).then(results => { const el = document.getElementById('news-weather-strip-line'); if (el) { el.innerHTML = results.filter(r => !r.error).map(w => `
${w.region} ${weatherEmoji(w.description)} ${w.temp || 23}°
`).join(''); } // 신문지면 미니 날씨 위젯 연동 const paperWeather = document.getElementById('paper-weather-widget-val'); if(paperWeather && results.length > 0) paperWeather.textContent = `${results[0].temp || 23}° ${weatherEmoji(results[0].description)}`; }); // 2. 검색 인풋 및 메뉴 태그 탭 전산 결합 활성화 const tagButtons = document.querySelectorAll('#news-tag-btn-group .tag-btn'); const searchInput = document.getElementById('news-inner-search-input'); tagButtons.forEach(btn => { btn.addEventListener('click', () => { tagButtons.forEach(b => b.classList.remove('active')); btn.classList.add('active'); if (searchInput) searchInput.value = ''; renderAllComponents(btn.textContent); }); }); if (searchInput) { searchInput.addEventListener('keydown', (e) => { if (e.key === 'Enter') { const val = searchInput.value.trim(); if (val.length > 0) renderAllComponents(val); } }); } // 3. 수동 새로고침 버튼 위임 연동 const manualRefreshBtn = document.getElementById('news-feed-manual-refresh-btn'); if(manualRefreshBtn) { manualRefreshBtn.addEventListener('click', () => { const activeText = document.querySelector('#news-tag-btn-group .tag-btn.active')?.textContent || '전체 뉴스'; renderAllComponents(activeText); }); } // 메인 오케스트레이션 1회차 호출 개시 await renderAllComponents('전체 뉴스'); }; return { init }; })();
